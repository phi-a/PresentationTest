\begin{frame}[t]{Dark matter}
  \Takeaway{We detect its gravity, but not its light.}

  \vspace{0.55cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \SectionLabel{Galaxy rotation}
      Outer stars move faster than visible matter alone predicts.
    \end{column}
    \begin{column}{0.48\textwidth}
      \SectionLabel{Gravitational lensing}
      Distorted background galaxies reveal mass we cannot see.
    \end{column}
  \end{columns}

  \vfill
  {\large\textbf{Open question:} What is dark matter made of?\par}
  \Source{\href{https://science.nasa.gov/dark-matter/}{NASA Science, ``Dark Matter''}}
\end{frame}
