\begin{frame}{Dark matter}
  \begin{itemize}
    \setlength{\itemsep}{0.24cm}
    \item Galaxy rotation curves show outer stars moving faster than visible matter alone predicts.
    \item Gravitational lensing reveals additional mass through distorted background galaxies.
    \item What makes up dark matter remains unknown.
  \end{itemize}

  \vfill
  \Source{\href{https://science.nasa.gov/dark-matter/}{NASA Science, ``Dark Matter''}}
\end{frame}
