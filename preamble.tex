\definecolor{illinoisblue}{HTML}{004C97}
\definecolor{bodygray}{HTML}{262626}
\definecolor{mutedgray}{HTML}{7F7F7F}
\setbeamercolor{background canvas}{bg=white}
\setbeamercolor{normal text}{fg=bodygray}
\setbeamercolor{frametitle}{fg=illinoisblue,bg=white}
\setbeamercolor{structure}{fg=illinoisblue}
\setbeamercolor{itemize item}{fg=illinoisblue}
\setbeamercolor{itemize subitem}{fg=illinoisblue}

\setbeamerfont{normal text}{family=\sffamily,size=\fontsize{18pt}{22pt}\selectfont}
\setbeamerfont{frametitle}{family=\sffamily,size=\fontsize{24pt}{29pt}\selectfont,series=\bfseries}
\setbeamerfont{itemize/enumerate body}{family=\sffamily,size=\fontsize{18pt}{22pt}\selectfont}
\setbeamerfont{itemize/enumerate subbody}{family=\sffamily,size=\fontsize{18pt}{22pt}\selectfont}

\setbeamertemplate{navigation symbols}{}
\setbeamersize{text margin left=0.6cm,text margin right=0.6cm}
\setlength{\parskip}{0pt}

% The reference master uses Fermilab and DarkNESS marks in the header.
% Figure lookup supports PNG and PDF assets in figures/.
\graphicspath{{figures/}}
\newcommand{\ResearchRule}{\textcolor{illinoisblue}{\rule{\dimexpr\paperwidth-1.2cm\relax}{1.25pt}}}
\newcommand{\ResearchRuleRow}{\hbox to \paperwidth{\hspace*{0.6cm}\ResearchRule\hss}}
\newcommand{\ResearchPageNumber}{%
  \hbox to \paperwidth{\hfill{\fontsize{10pt}{10pt}\selectfont\color{illinoisblue}\insertframenumber}\hspace*{0.6cm}}%
}

% Main-slide master: title and both logos, then a blue divider.
\setbeamertemplate{frametitle}{%
  \nointerlineskip
  \vbox to 0.86cm{%
    \vfil
    \hbox to \paperwidth{%
      \hspace*{0.6cm}%
      \begin{minipage}[c][0.62cm][c]{7.05cm}%
        {\usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}\insertframetitle}%
      \end{minipage}%
      \hspace*{0.22cm}%
      \includegraphics[width=1.97cm,height=0.42cm,keepaspectratio]{FERMI_LOGO-1.png}%
      \hspace*{0.22cm}%
      \includegraphics[width=2.04cm,height=0.50cm,keepaspectratio]{DarkNESS_LOGO-1.png}%
      \hfill
    }%
    \vfil
  }%
  \ResearchRuleRow
  \vspace*{0.196cm}%
}

% Main-slide footer: matching blue divider and blue page number.
\setbeamertemplate{footline}{%
  \vbox to 0.48cm{%
    \ResearchRuleRow
    \vspace*{0.04cm}%
    \ResearchPageNumber%
    \vfil
  }%
}
